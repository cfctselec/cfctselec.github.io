\question[3] Un chauffe-eau électrique d'une puissance de \qty{2200}{[\watt]} fonctionne à pleine puissance pendant une durée de \qty{3,5}{[\hour]} pour chauffer un réservoir d'eau. Calculer l'énergie électrique consommée par cet appareil en kilowattheures~?

\begin{solution}
	\begin{align*}
		P &= \qty{2200}{[\watt]} = \qty{2,2}{[\kilo\watt]} \\[6pt]
		t &= \qty{3,5}{[\hour]} \\[6pt]
		W &= P \cdot t \\[6pt]
		W &= \qty{2,2}{[\kilo\watt]} \cdot \qty{3,5}{[\hour]} \\[6pt]
		W &= \qty{7,7}{[\kilo\watt\hour]} \\[6pt]
	\end{align*}
	
	\begin{verbatim}
		% Télécharger OCTAVE depuis https://wiki.octave.org/Using_Octave et l'installer
		% Puis copier-coller le code dans OCTAVE
		
		% Données
		P = 2200;    % Puissance [W]
		t = 3.5;     % Temps [h]
		
		% Calcul
		W_kwh = (P / 1000) * t;
		
		% Affichage
		printf("Puissance de l'appareil : %.1f [kW]\n", P/1000);
		printf("Temps de fonctionnement : %.1f [h]\n", t);
		printf("Énergie consommée : %.2f [kWh]\n", W_kwh);
	\end{verbatim}
\end{solution}
\end{questions}
