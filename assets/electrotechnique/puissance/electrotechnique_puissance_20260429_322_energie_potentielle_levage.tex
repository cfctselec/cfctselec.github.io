\question[6] Un monte-charge doit élever une palette de \qty{650}{[\kilo\gram]} à une hauteur de \qty{12}{[\meter]}. En négligeant les frottements, calculer l'énergie mécanique (travail) nécessaire pour effectuer ce déplacement. On utilisera $g = \qty{9,81}{[\meter/\second^2]}$~?
\vspace{0.5em}
\needspace{55\baselineskip}
\begin{solution}
	\begin{align*}
		m &= \qty{650}{[\kilo\gram]} \\[6pt]
		g &= \qty{9,81}{[\meter/\second^2]} \\[6pt]
		h &= \qty{12}{[\meter]} \\[6pt]
		W_p &= m \cdot g \cdot h \\[6pt]
		W_p &= \qty{650}{[\kilo\gram]} \cdot \qty{9,81}{[\meter/\second^2]} \cdot \qty{12}{[\meter]} \\[6pt]
		W_p &= \qty{76518}{[\joule]} = \qty{76,52}{[\kilo\joule]} \\[6pt]
	\end{align*}
	
	\begin{verbatim}
		% Télécharger OCTAVE depuis https://wiki.octave.org/Using_Octave et l'installer
		% Puis copier-coller le code dans OCTAVE
		
		% Données
		m = 650;     % Masse [kg]
		g = 9.81;    % Gravité [m/s^2]
		h = 12;      % Hauteur [m]
		
		% Calcul
		Wp = m * g * h;
		
		% Affichage
		printf("Masse de la charge : %.0f [kg]\n", m);
		printf("Hauteur de levage : %.0f [m]\n", h);
		printf("Énergie potentielle (travail) : %.2f [J]\n", Wp);
		printf("Soit : %.2f [kJ]\n", Wp / 1000);
	\end{verbatim}
\end{solution}
\end{questions}
