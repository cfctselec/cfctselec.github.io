\section*{Préparation TS PQ CFC ELMO,IELE,PELE}
\begin{questions}
\question[6] Une pile d'une tension de \qty{4,5}{[V]} et d'une capacité de \qty{3}{[Ah]} coûte \qty{1,50}{CHF}.\\Quel est le prix en CHF de la conversion pour \qty{1}{[kWh]} d'énergie ?

\begin{solution}

Calcul de l'énergie totale emmagasinée par la pile ($W$) :
\begin{align*}
W &= U \cdot C \\[6pt]
W &= \qty{4,5}{[V]} \cdot \qty{3}{[Ah]} = \qty{13,5}{[Wh]}
\end{align*}

Conversion de l'énergie totale en kilowattheures ($W_{kWh}$) :
\begin{align*}
W_{kWh} &= \dfrac{W}{\qty{1000}{[Wh/kWh]}} \\[6pt]
W_{kWh} &= \dfrac{\qty{13,5}{[Wh]}}{\qty{1000}{[Wh/kWh]}} = \qty{0,0135}{[kWh]}
\end{align*}

Calcul du coût de l'énergie par kilowattheure ($C_{kWh}$) :
\begin{align*}
C_{kWh} &= \dfrac{\text{Prix}}{W_{kWh}} \\[6pt]
C_{kWh} &= \dfrac{\qty{1,50}{[CHF]}}{\qty{0,0135}{[kWh]}} = \qty{111,1111}{[CHF/kWh]}
\end{align*}

\begin{verbatim}
% Données de l'énoncé
U = 4.5;          % Tension de la pile [V]
C = 3;            % Capacité de la pile [Ah]
prix = 1.50;      % Prix d'achat de la pile [CHF]

% 1. Calcul de l'énergie totale en Wh (notion d'énergie notée W)
W_wh = U * C;
printf("Énergie totale de la pile : W_wh = %.2f [Wh]\n", W_wh);

% 2. Conversion de l'énergie en kWh
W_kwh = W_wh / 1000;
printf("Énergie totale en kWh : W_kwh = %.5f [kWh]\n", W_kwh);

% 3. Calcul du coût du kWh d'énergie
cout_kwh = prix / W_kwh;
printf("Coût du kWh d'énergie : cout_kwh = %.4f [CHF/kWh]\n", cout_kwh);
\end{verbatim}
\end{solution}
\end{questions}
