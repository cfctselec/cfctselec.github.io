\section*{Préparation TS PQ CFC ELMO,IELE,PELE}
\begin{questions}
\question[6] Calculez le flux magnétique à l'intérieur d'une bobine de \qty{25}{[cm]} de longueur, sur laquelle sont enroulées \qty{800}{[spires]} formant une section de \qty{15}{[cm^2]}, parcourues par un courant de \qty{5}{[A]}.

\begin{solution}
La bobine possède un noyau d'air, sa perméabilité relative est donc $\mu_r = 1$. La perméabilité du vide est définie par la constante $\mu_0 = 4\pi \cdot 10^{-7}{[\text{H/m}]}$.

Calcul de l'excitation magnétique (champ magnétique $H$) au centre de la bobine~:
\begin{align*}
H &= \dfrac{N \cdot I}{l} \\[6pt]
H &= \dfrac{800 \cdot \qty{5}{[A]}}{\qty{0,25}{[m]}} = \qty{16000}{[A/m]}
\end{align*}

Calcul de l'induction magnétique ($B$) :
\begin{align*}
B &= \mu_0 \cdot \mu_r \cdot H \\[6pt]
B &= 4\pi \cdot 10^{-7} \cdot 1 \cdot 16000 = \qty{0,020106}{[T]}
\end{align*}

Calcul du flux magnétique ($\Phi$) à travers la section transversale $S$ :
La section doit être convertie en mètres carrés : $S = \qty{15}{[cm^2]} = \qty{15e-4}{[m^2]}$.
\begin{align*}
\Phi &= B \cdot S \\[6pt]
\Phi &= \qty{0,020106}{[T]} \cdot \qty{15e-4}{[m^2]} = \qty{3,0159e-5}{[Wb]} = \qty{30,159}{[\mu Wb]}
\end{align*}

\begin{verbatim}
% Données de l'énoncé
N = 800;              % Nombre de spires
l = 0.25;             % Longueur de la bobine [m] (25 cm)
S = 15e-4;            % Section de la bobine [m^2] (15 cm^2)
I = 5;                % Intensité du courant [A]
mu_0 = 4 * pi * 1e-7; % Perméabilité du vide [H/m]

% 1. Calcul du champ magnétique H
H = (N * I) / l;
printf("Champ magnétique : H = %.2f [A/m]\n", H);

% 2. Calcul de l'induction magnétique B
B = mu_0 * H;
printf("Induction magnétique : B = %.6f [T]\n", B);

% 3. Calcul du flux magnétique Phi
Phi = B * S;
printf("Flux magnétique : Phi = %.8e [Wb] (soit %.3f [uWb])\n", Phi, Phi * 1e6);
\end{verbatim}
\end{solution}
\end{questions}
