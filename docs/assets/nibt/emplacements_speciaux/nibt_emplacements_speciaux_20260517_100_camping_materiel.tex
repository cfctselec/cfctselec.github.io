\section*{Préparation DT PQ CFC ELMO,IELE,PELE}
\begin{questions}
\question
Soit l'infrastructure électrique d'un nouveau camping. Vous devez sélectionner le matériel de raccordement pour les emplacements de caravanes.
\begin{parts}
	\part[3] Quel type de socle de prise est impérativement requis pour l'alimentation des véhicules de loisir~?
	\part[3] Quel est le degré de protection IP minimal que doivent présenter ces socles de prises lorsqu'ils sont installés à l'air libre~?
	\part[3] Quelles sont les limites de hauteur (minimale et maximale) à respecter pour la pose des socles de prises par rapport au sol fini~?
	\part[3] Quel est le nombre maximal de socles de prises autorisé par point d'alimentation (distributeur)~?
\end{parts}

\begin{solution}
	\begin{itemize}
		\item \textbf{Type de prise} : Le socle doit être de type EN 60309-2 (modèle CEE 16 A LNPE bleu). Les prises domestiques de type 13 ou 23 sont interdites pour cet usage (NIBT 7.08.5.5.1.6).
		\item \textbf{Indice de protection} : Le matériel doit être adapté aux projections d'eau (AD4), soit un degré IP 44 au minimum (NIBT 7.08.5.1.2.1).
		\item \textbf{Hauteur de pose} : Les prises doivent se situer à une hauteur comprise entre \qty{0.5}{[m]} et \qty{1.5}{[m]} du sol (NIBT 7.08.5.5.1.6).
		\item \textbf{Distribution} : Un distributeur ne peut pas alimenter plus de 4 emplacements (maximum 4 prises par point d'alimentation) (NIBT 7.08.5.5.1.3).
	\end{itemize}
	
	\begin{align*}
		L_{\text{type\_conforme}} &= \text{EN 60309-2 (CEE)} \\[6pt]
		IP_{\text{min}} &= \text{IP 44} \\[6pt]
		H_{\text{pose}} &\in [\qty{0.5}{[m]} ; \qty{1.5}{[m]}] \\[6pt]
		N_{\text{prises\_max}} &= 4 \\[6pt]
		I_{\Delta n} &\leq \qty{30}{[mA]} \text{ (Protection individuelle par socle)}
	\end{align*}
\end{solution}
\end{questions}
