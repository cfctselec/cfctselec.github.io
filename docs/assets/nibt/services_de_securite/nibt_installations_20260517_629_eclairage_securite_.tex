\section*{Préparation DT PQ CFC ELMO,IELE,PELE}
\begin{questions}
\question
En tant que responsable technique pour la rénovation d'un bâtiment administratif du village de Baulmes, vous devez valider la conformité de l'éclairage de sécurité pour les voies d'évacuation. 

\begin{parts}
	\part[1] Quel est l'éclairement minimal exigé au sol pour une voie d'évacuation menant à un lieu sûr~?
	\part[1] Dans quel chapitre spécifique de la NIBT trouvez-vous les prescriptions générales concernant les sources d'alimentation pour les services de sécurité~?
	\part[1] Pour garantir le maintien de fonction en cas de sinistre, citez une exigence de pose concernant le cheminement des circuits de sécurité par rapport aux circuits normaux.
\end{parts}

\begin{solution}
	\begin{itemize}
		\item \textbf{Éclairement minimal} : L'éclairement au sol sur l'axe central de la voie d'évacuation doit être au minimum de \qty{1}{[\lux]}.
		\item \textbf{Référence normative} : Les services de sécurité se trouvent dans le chapitre NIBT \num{5.6}.
		\item \textbf{Maintien de fonction} : Les circuits doivent être séparés des autres circuits par des matériaux résistant au feu, ou suivre des cheminements différents, ou posséder une enveloppe résistante au feu (NIBT 5.6.7.1).
	\end{itemize}
	

	\end{solution}
\end{questions}
