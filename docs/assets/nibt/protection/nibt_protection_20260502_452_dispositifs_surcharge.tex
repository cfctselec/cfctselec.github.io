\section*{Préparation DT PQ CFC ELMO,IELE,PELE}
\begin{questions}
\question
Indiquez pour chaque affirmation concernant les dispositifs de protection s'il s'agit d'un matériel assurant la fonction de protection contre les surcharges.

\begin{parts}
	\part[1] Le dispositif de protection à courant différentiel-résiduel (RCD) assure la protection uniquement contre les surcharges.
	\part[1] Le fusible HPC assure la protection contre les surcharges.
	\part[1] Le relais de protection moteur assure la protection contre les surcharges.
	\part[1] Le fusible miniaturisé assure la protection contre les surcharges.
\end{parts}

\begin{solution}
	NIBT 4.3.2.2 Les dispositifs de protection contre les surcharges doivent être capables d'interrompre tout courant de surcharge dans les conducteurs avant qu'un tel courant ne puisse provoquer un échauffement nuisible aux isolations, aux connexions, aux canalisations ou à l'environnement.
	
	\begin{itemize}
		\item \textbf{RCD (DDR) :} Ce matériel est conçu pour la protection contre les courants de défaut (protection des personnes) et ne remplit pas la fonction de protection contre les surcharges.\\ L'affirmation est donc \textbf{Fausse}.
		\item \textbf{Fusible HPC :} Les fusibles à Haut Pouvoir de Coupure assurent la protection contre les surintensités, ce qui inclut les courants de surcharge et de court-circuit.\\
		L'affirmation est \textbf{Juste}.
		\item \textbf{Relais de protection moteur :} Il s'agit d'un dispositif thermique spécifiquement utilisé pour la protection des récepteurs contre les surcharges prolongées.\\
		L'affirmation est \textbf{Juste}.
		\item \textbf{Fusible miniaturisé :} Ce type de coupe-surintensité assure la protection contre les surcharges dans les circuits de faible puissance.\\
		L'affirmation est \textbf{Juste}.
	\end{itemize}
\end{solution}
\end{questions}
