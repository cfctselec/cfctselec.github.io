\section*{Préparation TS PQ CFC IELE,PELE}
\begin{questions}
\question[9]
En analysant le marquage technique présent sur le châssis de la prise de référence (type Kathrein ESD 33), déterminez les trois caractéristiques essentielles suivantes nécessaires pour les calculs
\begin{center}
 	\includegraphics[scale=0.13]{ESD33.jpg}
\end{center}

\begin{parts}
    \part L'atténuation subie par le signal lorsqu'il traverse la prise pour alimenter l'étage inférieur.
    \part L'atténuation subie par le signal entre l'entrée de la prise et la sortie raccordée au récepteur de l'utilisateur (TV/Radio).
    \part La plage de fréquence totale supportée par cet équipement pour assurer la compatibilité avec les signaux SAT.
\end{parts}
\begin{solution}

\begin{itemize}
    \item Atténuation de passage : \qty{1.9}{[\decibel]}. (C'est la perte d'insertion entre les bornes "IN" et "OUT" de la prise).
    \item Atténuation de raccordement (ou connexion) : \qty{14}{[\decibel]}. 
    \item  Plage de fréquence : \qtyrange{5}{2150}{[\mega\hertz]}.
    \item Il faut ouvrir tous les pdf des examens de TC \url{https://qrhd1.github.io/oldcfc/}  où il y a le genre de problème ci-dessous \href{https://raw.githubusercontent.com/qrhd1/oldcfc/main/pdf_files/2022_TC_IEL_corrige.pdf}{2022\_TC\_IEL\_corrige.pdf}.


\end{itemize}
\begin{center}
	\includegraphics[scale=0.65]{2022_TC_IEL_EXTRAIT.png}
\end{center}

\end{solution}
\end{questions}
