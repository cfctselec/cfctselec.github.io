	\begin{questions}

\question[6] Quelle quantité d'énergie thermique $Q$ est nécessaire pour chauffer \qty{2,5}{[\kilo\gram]} d'eau de \qty{15}{[\celsius]} à \qty{85}{[\celsius]}~?\\
La chaleur massique de l'eau est $c = \qty{4186}{[\joule/(\kilo\gram\cdot\kelvin)]}$.

\begin{solution}
	\begin{align*}
		m &= \qty{2,5}{[\kilo\gram]} \\[6pt]
		c &= \qty{4186}{[\joule/(\kilo\gram\cdot\kelvin)]} \\[6pt]
		\Delta \theta &= 85 - 15 = \qty{70}{[\celsius]} \\[6pt]
		Q &= m \cdot c \cdot \Delta \theta \\[6pt]
		Q &= \qty{2,5}{[\kilo\gram]} \cdot \qty{4186}{[\joule/(\kilo\gram\cdot\kelvin)]} \cdot \qty{70}{[\celsius]} \\[6pt]
		Q &= \qty{732550}{[\joule]} = \qty{732,55}{[\kilo\joule]} \\[6pt]
	\end{align*}
	
	\begin{verbatim}
		% Télécharger OCTAVE depuis https://wiki.octave.org/Using_Octave et l'installer
		% Puis copier-coller le code dans OCTAVE
		
		% Données
		m = 2.5;      % Masse [kg]
		c = 4186;     % Capacité thermique massique [J/kgK]
		T1 = 15;      % Temp initiale [°C]
		T2 = 85;      % Temp finale [°C]
		
		% Calcul
		delta_T = T2 - T1;
		Q = m * c * delta_T;
		
		% Affichage
		printf("Différence de température : %.0f [K]\n", delta_T);
		printf("Énergie thermique nécessaire : %.0f [J]\n", Q);
		printf("Soit : %.2f [kJ]\n", Q / 1000);
	\end{verbatim}
\end{solution}
\end{questions}
