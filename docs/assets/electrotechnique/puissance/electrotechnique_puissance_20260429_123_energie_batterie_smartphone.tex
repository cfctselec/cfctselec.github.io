\question La batterie d'un smartphone récent possède une capacité de \qty{4500}{[\milli\ampere\hour]} sous une tension nominale de \qty{3,8}{[\volt]}.
\begin{parts}
	\part[3] Calculer l'énergie stockée dans cette batterie en wattheures~?
	\part[3] Convertir cette énergie en joules~?
\end{parts}

\begin{solution}
	\begin{align*}
		\text{a) Calcul de l'énergie en wattheures (Wh) :} & \\[6pt]
		Q &= \qty{4500}{[\milli\ampere\hour]} = \qty{4,5}{[\ampere\hour]} \\[6pt]
		U &= \qty{3,8}{[\volt]} \\[6pt]
		W &= U \cdot Q = \qty{3,8}{[\volt]} \cdot \qty{4,5}{[\ampere\hour]} \\[6pt]
		W &= \qty{17,1}{[\watt\hour]} \\[6pt]
		\text{b) Conversion en joules (J) :} & \\[6pt]
		\text{Sachant que } \qty{1}{[\watt\hour]} &= \qty{3600}{[\joule]} \\[6pt]
		W_{joules} &= W \cdot 3600 = \qty{17,1}{[\watt\hour]} \cdot 3600 \\[6pt]
		W_{joules} &= \qty{61560}{[\joule]} \\[6pt]
	\end{align*}
	
	\begin{verbatim}
		% Télécharger OCTAVE depuis https://wiki.octave.org/Using_Octave et l'installer
		% Puis copier-coller le code dans OCTAVE
		
		% Données
		Q_mah = 4500;    % Capacité en mAh
		U = 3.8;         % Tension nominale [V]
		
		% Calculs
		Q_ah = Q_mah / 1000;
		W_wh = U * Q_ah;
		W_j = W_wh * 3600;
		
		% Affichage
		printf("Capacité : %.1f [Ah]\n", Q_ah);
		printf("Énergie stockée : %.2f [Wh]\n", W_wh);
		printf("Énergie stockée : %.0f [J]\n", W_j);
	\end{verbatim}
\end{solution}
\end{questions}
